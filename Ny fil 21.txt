
using UdonSharp;
using UnityEngine;
using VRC.SDK3.Components;
using VRC.SDKBase;
using VRC.Udon;

public class ButtonMove : UdonSharpBehaviour
{
    public RectTransform button;
    public override void Interact()
    {
        button.anchoredPosition = new Vector2(-40, 0);
        button.TweenAnchorPos(Vector3.zero, 1f, VRCTweenEase.OutCubic)
            .OnComplete(this, nameof(OnScaleComplete));
    }

    public void OnScaleComplete()
    {
        Debug.Log("Scale animation complete!");
    }
}
